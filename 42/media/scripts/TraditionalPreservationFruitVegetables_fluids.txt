module TraditionalPreservationFruitVegetables
{
    /* Tomato paste in a pot or TP Core's cauldron (TraditionalPreservationFruitVegetablesPaste.lua): each tomato goes
       in as 0.15 L of TomatoPulp; kept steaming 3 hours it is TomatoPaste, packed into a crock. Pulp left raw goes off
       when its oldest tomato would have, paste left unpacked a week after it was cooked (RottenTomatoPulp). Pulp and
       paste have no category: the game offers Drink only for a Beverage, and Hazardous would mark good paste as
       dangerous in its tooltip (user, 2026-11-27). The rotten pulp is Hazardous. Never rename them once the mod is out: saves hold the
       names. */
    fluid TomatoPulp
    {
        ColorReference = Tomato,
        DisplayName = Fluid_Name_TomatoPulp,
    }

    fluid TomatoPaste
    {
        ColorReference = Brown,
        DisplayName = Fluid_Name_TomatoPaste,
    }

    fluid RottenTomatoPulp
    {
        ColorReference = DarkOliveGreen,
        DisplayName = Fluid_Name_RottenTomatoPulp,
        Categories
        {
            Hazardous,
        }
        Properties
        {
            UnhappyChange = 60.0,
        }
        Poison
        {
            maxEffect = Mild,
            minAmount = 0.1,
            diluteRatio = 0.2,
        }
    }

    /* Kraut brine (TraditionalPreservationFruitVegetablesBarrel.lua): 0.15 L a head packed, drawn off a ready barrel as a
       starter (1 L at most while kraut is in it) or drained once the kraut is out. Good brine is a Beverage (sour and
       salty, drunk by the glass); on TP Core's FluidAgeing it goes mouldy in about 3 weeks at 20 C (RottenKrautBrine,
       Hazardous). Values are a litre's. Its colour is a cloudy pale grey-green, apart from Salt's Brine (Khaki) and the
       mouldy brine (DarkOliveGreen). */
    fluid KrautBrine
    {
        ColorReference = DarkSeaGreen,
        DisplayName = Fluid_Name_KrautBrine,
        Categories
        {
            Beverage,
        }
        Properties
        {
            ThirstChange = -10.0,
            UnhappyChange = 5.0,
            Calories = 40.0,
            Carbohydrates = 8.0,
            Proteins = 1.0,
        }
    }

    fluid RottenKrautBrine
    {
        ColorReference = DarkOliveGreen,
        DisplayName = Fluid_Name_RottenKrautBrine,
        Categories
        {
            Hazardous,
        }
        Properties
        {
            UnhappyChange = 40.0,
        }
        Poison
        {
            maxEffect = Mild,
            minAmount = 0.1,
            diluteRatio = 0.2,
        }
    }
}

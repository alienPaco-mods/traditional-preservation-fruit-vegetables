module TraditionalPreservationFruitVegetables
{

       dangerous in its tooltip (user, 2026-11-27). The rotten pulp is Hazardous. Never rename them once the mod is out: saves hold the
       names. */

    {
        ColorReference = Tomato,
        DisplayName = Fluid_Name_TomatoPulp,
    }

    fluid TomatoPaste
    {
        ColorReference = Brown,
        DisplayName = Fluid_Name_TomatoPaste,
    }

    {
        ColorReference = DarkOliveGreen,
        DisplayName = Fluid_Name_RottenTomatoPulp,
        Categories
        {
            Hazardous,
        }
        Properties
        {
            UnhappyChange = 60.0,
        }
        Poison
        {
            maxEffect = Mild,
            minAmount = 0.1,
            diluteRatio = 0.2,
        }
    }
